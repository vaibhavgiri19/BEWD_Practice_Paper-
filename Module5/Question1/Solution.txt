// app.js
// ----------------------
// 🏥 Hospital Management System (Beginner Friendly)
// Secure Staff Login using JWT & HTTP-only Cookies
// ----------------------

const express = require("express");
const bcrypt = require("bcryptjs");
const jwt = require("jsonwebtoken");
const cookieParser = require("cookie-parser");

const app = express();
app.use(express.json());
app.use(cookieParser());

// Temporary in-memory storage (simulate database)
const staffList = [];
const patients = [];

// Secret key for JWT
const JWT_SECRET = "my_hospital_secret_key";

// ----------------------
// 1️⃣ Register Route
// ----------------------
app.post("/api/staff/register", async (req, res) => {
  const { name, email, password } = req.body;

  if (!name || !email || !password) {
    return res.status(400).json({ success: false, error: "All fields are required" });
  }

  // Check if email already exists
  const existingStaff = staffList.find((s) => s.email === email);
  if (existingStaff) {
    return res.status(400).json({ success: false, error: "Staff already registered" });
  }

  // Hash password before saving
  const hashedPassword = await bcrypt.hash(password, 10);

  const newStaff = {
    id: staffList.length + 1,
    name,
    email,
    password: hashedPassword,
  };
  staffList.push(newStaff);

  res.json({ success: true, message: "Staff registered successfully", staffId: newStaff.id });
});

// ----------------------
// 2️⃣ Login Route
// ----------------------
app.post("/api/staff/login", async (req, res) => {
  const { email, password } = req.body;

  if (!email) return res.status(400).json({ success: false, error: "Email is required" });
  if (!password) return res.status(400).json({ success: false, error: "Password is required" });

  const staff = staffList.find((s) => s.email === email);
  if (!staff) return res.status(400).json({ success: false, error: "Staff not found" });

  // Compare passwords
  const isMatch = await bcrypt.compare(password, staff.password);
  if (!isMatch) return res.status(401).json({ success: false, error: "Invalid password" });

  // Generate JWT token
  const token = jwt.sign({ id: staff.id, email: staff.email }, JWT_SECRET, { expiresIn: "1h" });

  // Set token in HTTP-only cookie
  res.cookie("staff_token", token, {
    httpOnly: true,
    sameSite: "strict",
    secure: false, // set true in production (HTTPS)
    maxAge: 3600000,
  });

  res.json({ success: true, message: "Staff login successful", staff: staff.name });
});

// ----------------------
// 3️⃣ Middleware - Verify Token
// ----------------------
function authMiddleware(req, res, next) {
  const token = req.cookies.staff_token;

  if (!token) {
    return res.status(401).json({ success: false, error: "Access denied. Please log in." });
  }

  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    req.staff = decoded; // store staff info in req
    next();
  } catch (err) {
    return res.status(403).json({ success: false, error: "Invalid or expired token" });
  }
}

// ----------------------
// 4️⃣ Protected CRUD Routes for Patients
// ----------------------

// ➕ Create patient
app.post("/api/patients", authMiddleware, (req, res) => {
  const { name, age, diagnosis } = req.body;

  if (!name || !age || !diagnosis) {
    return res.status(400).json({ success: false, error: "All patient fields are required" });
  }

  const newPatient = {
    id: patients.length + 1,
    name,
    age,
    diagnosis,
    createdBy: req.staff.email,
  };

  patients.push(newPatient);
  res.json({ success: true, message: "Patient added successfully", patientId: newPatient.id });
});

// 📋 Get all patients (for logged-in staff only)
app.get("/api/patients", authMiddleware, (req, res) => {
  const staffPatients = patients.filter((p) => p.createdBy === req.staff.email);
  res.json({ success: true, data: staffPatients });
});

// ✏️ Update patient
app.put("/api/patients/:id", authMiddleware, (req, res) => {
  const id = parseInt(req.params.id);
  const patient = patients.find((p) => p.id === id);

  if (!patient) return res.status(404).json({ success: false, error: "Patient not found" });
  if (patient.createdBy !== req.staff.email)
    return res.status(403).json({ success: false, error: "Unauthorized to edit this record" });

  const { name, age, diagnosis } = req.body;
  if (name) patient.name = name;
  if (age) patient.age = age;
  if (diagnosis) patient.diagnosis = diagnosis;

  res.json({ success: true, message: "Patient updated successfully" });
});

// ❌ Delete patient
app.delete("/api/patients/:id", authMiddleware, (req, res) => {
  const id = parseInt(req.params.id);
  const index = patients.findIndex((p) => p.id === id);

  if (index === -1) return res.status(404).json({ success: false, error: "Patient not found" });
  if (patients[index].createdBy !== req.staff.email)
    return res.status(403).json({ success: false, error: "Unauthorized to delete this record" });

  patients.splice(index, 1);
  res.json({ success: true, message: "Patient deleted successfully" });
});

// ----------------------
// 5️⃣ Logout Route
// ----------------------
app.post("/api/staff/logout", (req, res) => {
  res.clearCookie("staff_token");
  res.json({ success: true, message: "Logged out successfully" });
});

// ----------------------
// 6️⃣ Start Server
// ----------------------
app.listen(3000, () => {
  console.log("✅ Server running on http://localhost:3000");
});
