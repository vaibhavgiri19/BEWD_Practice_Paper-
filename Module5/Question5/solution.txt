const express = require("express");
const bcrypt = require("bcryptjs");
const jwt = require("jsonwebtoken");
const cookieParser = require("cookie-parser");

const app = express();
app.use(express.json());
app.use(cookieParser());

const JWT_SECRET = "ecommerce_secret_key";
const customers = [];

// Register customer
app.post("/api/customers/register", async (req, res) => {
  const { name, email, password } = req.body;
  if (!name || !email || !password)
    return res.status(400).json({ success: false, error: "All fields required" });

  const hashed = await bcrypt.hash(password, 10);
  customers.push({ name, email, password: hashed });
  res.json({ success: true, message: "Customer registered successfully" });
});

// Login customer
app.post("/api/customers/login", async (req, res) => {
  const { email, password } = req.body;
  const customer = customers.find((c) => c.email === email);
  if (!customer) return res.status(401).json({ success: false, error: "Invalid credentials" });

  const valid = await bcrypt.compare(password, customer.password);
  if (!valid) return res.status(401).json({ success: false, error: "Invalid password" });

  const token = jwt.sign({ email: customer.email }, JWT_SECRET, { expiresIn: "1h" });
  res.cookie("token", token, { httpOnly: true, sameSite: "strict", secure: false, maxAge: 3600000 });
  res.json({ success: true, message: "Customer login successful", customer: customer.name });
});

// Protected route
app.get("/api/orders", (req, res) => {
  const token = req.cookies.token;
  if (!token) return res.status(403).json({ success: false, error: "Access denied" });
  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    res.json({ success: true, message: "Access granted to orders", user: decoded.email });
  } catch {
    res.status(403).json({ success: false, error: "Invalid or expired token" });
  }
});

// Logout
app.post("/api/customers/logout", (req, res) => {
  res.clearCookie("token");
  res.json({ success: true, message: "Logged out successfully" });
});

app.listen(3004, () => console.log("E-Commerce server running on port 3004"));
